364public:
365 const T& getValue(const State& s, int derivOrder) const {
366 SimTK_ERRCHK2(0 <= derivOrder && derivOrder <= getNumTimeDerivatives(),
367 "Measure_<T>::getValue()",
368 "derivOrder %d was out of range; this Measure allows 0-%d.",
369 derivOrder, getNumTimeDerivatives());
370
371 // We require the stage to have been advanced to at least the one
372 // before this measure's depends-on stage since this will get called
373 // towards the end of the depends-on stage realization.
374 if (getDependsOnStage(derivOrder) != Stage::Empty) {
375#ifndef NDEBUG
376 Stage prevStage = getDependsOnStage(derivOrder).prev();
377#endif
378
379 SimTK_ERRCHK2
380 ( ( isInSubsystem() && getStage(s)>=prevStage)
381 || (!isInSubsystem() && s.getSystemStage()>=prevStage),
382 "Measure_<T>::getValue()",
383 "Expected State to have been realized to at least stage "
384 "%s but stage was %s.",
385 prevStage.getName().c_str(),
386 (isInSubsystem() ? getStage(s) : s.getSystemStage())
387 .getName().c_str());
388 }
389
390 if (derivOrder < getNumCacheEntries()) {
391 if (!isCacheValueRealized(s,derivOrder)) {
392 T& value = updCacheEntry(s,derivOrder);
393 calcCachedValueVirtual(s, derivOrder, value);
394 markCacheValueRealized(s,derivOrder);
395 return value;
396 }
397 return getCacheEntry(s,derivOrder);
398 }
399
400 // We can't handle it here -- punt to the concrete Measure
401 // for higher order derivatives.
402 return getUncachedValueVirtual(s,derivOrder);
403 }
404

562private:
563 // Satisfy the realizeTopology() pure virtual here now that we know the
564 // data type T. Allocate lazy- or auto-validated- cache entries depending
565 // on the setting of presumeValidAtDependsOnStage.
566 void realizeTopology(State& s) const override final {
567 // Allocate cache entries. Initialize the value cache entry to
568 // the given defaultValue; all the derivative cache entries should be
569 // initialized to a NaN of the same size.
570 if (getNumCacheEntries()) {
571 derivIx[0] = presumeValidAtDependsOnStage
572 ? this->getSubsystem().allocateCacheEntry
573 (s, getDependsOnStage(0), new Value<T>(defaultValue))
574 : this->getSubsystem().allocateLazyCacheEntry
575 (s, getDependsOnStage(0), new Value<T>(defaultValue));
576
577 if (getNumCacheEntries() > 1) {
578 T nanValue; Measure_Num<T>::makeNaNLike(defaultValue, nanValue);
579 for (int i=1; i < getNumCacheEntries(); ++i) {
580 derivIx[i] = presumeValidAtDependsOnStage
581 ? this->getSubsystem().allocateCacheEntry
582 (s, getDependsOnStage(i), new Value<T>(nanValue))
583 : this->getSubsystem().allocateLazyCacheEntry
584 (s, getDependsOnStage(i), new Value<T>(nanValue));
585 }
586 }
587 }
588
589 // Call the concrete class virtual if any.
590 realizeMeasureTopologyVirtual(s);
591 }

1382

1396 const Subsystem& subsys = this->getSubsystem();
1397 if (subsys.isDiscreteVarUpdateValueRealized(s,resultIx))
1398 return;
1399
1400 const Real t0 = subsys.getDiscreteVarLastUpdateTime(s,resultIx);
1401 const Result& prevResult = Value<Result>::downcast
1402 (subsys.getDiscreteVariable(s,resultIx));
1403 const T& f0 = prevResult.operand;
1404 const T& fdot0 = prevResult.operandDot; // may be invalid
1405 const bool good0 = prevResult.derivIsGood;
1406
1407 const Real t = s.getTime();
1408 Result& result = Value<Result>::updDowncast
1409 (subsys.updDiscreteVarUpdateValue(s,resultIx));
1410 T& f = result.operand; // renaming
1411 T& fdot = result.operandDot;
1412 bool& good = result.derivIsGood;
1413
1414 f = operand.getValue(s);
1415 good = false;
1416 if (!isFinite(t0))
1417 fdot = this->getValueZero();
1418 else if (t == t0) {
1419 fdot = fdot0;
1420 good = good0;
1421 } else {
1422 fdot = (f-f0)/(t-t0); // 1st order
1423 if (good0)
1424 fdot = Real(2)*fdot - fdot0; // now 2nd order
1425 good = true; // either 1st or 2nd order estimate
1426 }
1427 subsys.markDiscreteVarUpdateValueRealized(s,resultIx);
1428 }

1432 bool forceUseApprox;
1433
1434 // TOPOLOGY CACHE
1435 mutable bool isApproxInUse;
1436 mutable DiscreteVariableIndex resultIx; // auto-update
1437};

1597 bool ensureExtremeHasBeenUpdated(const State& s) const {
1598 assert(extremeIx.isValid() && isNewExtremeIx.isValid());
1599 const Subsystem& subsys = this->getSubsystem();
1600
1601 // We may have already determined whether we're at a new extreme in
1602 // which case we don't need to do it again.
1603 if (subsys.isDiscreteVarUpdateValueRealized(s, isNewExtremeIx))
1604 return Value<bool>::downcast
1605 (subsys.getDiscreteVarUpdateValue(s,isNewExtremeIx));
1606
1607 // We're going to have to decide if we're at a new extreme, and if
1608 // so record the new extreme value in the auto-update cache entry of
1609 // the extreme value state variable.
1610
1611 // Get the previous extreme value and the current operand value.
1612 const T& prevExtreme = Value<T>::downcast
1613 (subsys.getDiscreteVariable(s,extremeIx));
1614 const T& currentVal = operand.getValue(s);
1615
1616 // Search to see if any element has reached a new extreme.
1617 bool foundNewExt = false;
1618 for (int i=0; i < this->size() && !foundNewExt; ++i)
1619 foundNewExt = isNewExtreme(Measure_Num<T>::get(currentVal,i),
1620 Measure_Num<T>::get(prevExtreme,i));
1621
1622 // Record the result and mark the auto-update cache entry valid
1623 // so we won't have to recalculate. When the integrator advances to the
1624 // next step this cache entry will be swapped with the corresponding
1625 // state and marked invalid so we'll be sure to recalculate each step.
1626 Value<bool>::updDowncast
1627 (subsys.updDiscreteVarUpdateValue(s,isNewExtremeIx)) = foundNewExt;
1628 subsys.markDiscreteVarUpdateValueRealized(s,isNewExtremeIx);
1629
1630 // Don't update the auto-update cache entry if we didn't see a new
1631 // extreme. That way no auto-update will occur and the state variable
1632 // will remain unchanged with the existing update time preserved.
1633 if (!foundNewExt)
1634 return false;
1635
1636 // We have encountered a new extreme. We'll record the new extreme
1637 // in the auto-update cache entry which will be used as the current
1638 // result until the integrator advances to the next step at which time
1639 // this will be swapped with the state variable to serve as the previous
1640 // extreme value until a further extreme is encountered.
1641 T& newExtreme = Value<T>::updDowncast
1642 (subsys.updDiscreteVarUpdateValue(s,extremeIx));
1643
1644 for (int i=0; i < this->size(); ++i)
1645 Measure_Num<T>::upd(newExtreme,i) =
1646 extremeOf(Measure_Num<T>::get(currentVal,i),
1647 Measure_Num<T>::get(prevExtreme,i));
1648
1649 // Marking this valid is what ensures that an auto-update occurs later.
1650 subsys.markDiscreteVarUpdateValueRealized(s,extremeIx);
1651 return true;
1652 }
1653private:
1654 // Return true if newVal is "more extreme" than oldExtreme, according
1655 // to the operation we're performing.
1656 bool isNewExtreme(const typename Measure_Num<T>::Element& newVal,
1657 const typename Measure_Num<T>::Element& oldExtreme) const
1658 {
1659 switch (operation) {
1660 case Extreme::Maximum: return newVal > oldExtreme;
1661 case Extreme::Minimum: return newVal < oldExtreme;
1662 case Extreme::MaxAbs: return std::abs(newVal) > std::abs(oldExtreme);
1663 case Extreme::MinAbs: return std::abs(newVal) < std::abs(oldExtreme);
1664 };
1665 SimTK_ASSERT1_ALWAYS(!"recognized",
1666 "Measure::Extreme::Implementation::isNewExtreme(): "
1667 "unrecognized operation %d", (int)operation);
1668 return false; /*NOTREACHED*/
1669 }
1670
1671 // Given the value of one element of the operand, and that value's time
1672 // derivative, determine whether the derivative is moving the element
1673 // into the "more extreme" direction, according to the operation.
1674 bool isExtremeDir(const typename Measure_Num<T>::Element& value,
1675 const typename Measure_Num<T>::Element& deriv) const
1676 {
1677 const int sv = sign(value), sd = sign(deriv);
1678 if (sd == 0) return false; // derivative is zero; not changing
1679 switch (operation) {
1680 case Extreme::Maximum: return sd == 1; // getting larger
1681 case Extreme::Minimum: return sd == -1; // getting smaller
1682 case Extreme::MaxAbs: return sv==0 || sd==sv; // abs is growing
1683 case Extreme::MinAbs: return sd == -sv;
1684 };
1685 SimTK_ASSERT1_ALWAYS(!"recognized",
1686 "Measure::Extreme::Implementation::isExtremeDir(): "
1687 "unrecognized operation %d", (int)operation);
1688 return false; /*NOTREACHED*/
1689 }
1690
1691 typename Measure_Num<T>::Element
1692 extremeOf(const typename Measure_Num<T>::Element& newVal,
1693 const typename Measure_Num<T>::Element& oldExtreme) const
1694 {
1695 return isNewExtreme(newVal,oldExtreme) ? newVal : oldExtreme;
1696 }

1698 // TOPOLOGY STATE
1699 Measure_<T> operand;
1700 Operation operation;
1701
1702 // TOPOLOGY CACHE
1703 mutable DiscreteVariableIndex extremeIx; // extreme so far; auto-update
1704
1705 // This auto-update flag records whether the current value is a new
1706 // extreme. We don't really need to save it as a state variable since you
1707 // can figure this out from the timestamp, but we need to to get invalidated
1708 // by the auto-update swap so that we'll figure it out anew each step.
1709 mutable DiscreteVariableIndex isNewExtremeIx;
1710};

// Hide from Doxygen.
1718// This helper class is the contents of the discrete state variable and
1719// corresponding cache entry maintained by this measure. The variable is
1720// auto-update, meaning the value of the cache entry replaces the state
1721// variable at the start of each step.
1722//
1723// Circular buffers look like this:
1724//
1725// oldest=0, n=0
1726// v
1727// Empty buffer: | available |
1728//
1729// By convention, oldest=0 whenever the buffer is empty.
1730//
1731//
1732// oldest next=(oldest+n)%capacity
1733// v v
1734// | available | | | | | | | available |
1735// ^ n=6 ^
1736// 0 capacity
1737// v v
1738// or | | | | | | available | | | | | | | | n=12
1739// ^ ^
1740// next oldest
1741// = (oldest+n)%capacity
1742//
1743// Number of entries = n (called size() below)
1744// Empty = n==0
1745// Full = n==capacity()
1746// Next available = (oldest+n)%capacity()
1747template <class T>
1748class Measure_Delay_Buffer {
1749public:
1750 explicit Measure_Delay_Buffer() {initDataMembers();}
1751 void clear() {initDataMembers();}
1752 int size() const {return m_size;} // # saved entries, *not* size of arrays
1753 int capacity() const {return m_times.size();}
1754 bool empty() const {return size()==0;}
1755 bool full() const {return size()==capacity();}
1756
1757 double getEntryTime(int i) const
1758 { assert(i < size()); return m_times[getArrayIndex(i)];}
1759 const T& getEntryValue(int i) const
1760 { assert(i < size()); return m_values[getArrayIndex(i)];}
1761
1762 enum {
1763 InitialAllocation = 8, // smallest allocation
1764 GrowthFactor = 2, // how fast to grow (double)
1765 MaxShrinkProofSize = 16, // won't shrink unless bigger
1766 TooBigFactor = 5 // 5X too much->maybe shrink
1767 };
1768
1769 // Add a new entry to the end of the list, throwing out old entries that
1770 // aren't needed to answer requests at tEarliest or later.
1771 void append(double tEarliest, double tNow, const T& valueNow) {
1772 forgetEntriesMuchOlderThan(tEarliest);
1773 removeEntriesLaterOrEq(tNow);
1774 if (full())
1775 makeMoreRoom();
1776 else if (capacity() > std::max((int)MaxShrinkProofSize,
1777 (int)TooBigFactor * (size()+1)))
1778 makeLessRoom(); // less than 1/TooBigFactor full
1779 const int nextFree = getArrayIndex(m_size++);
1780 m_times[nextFree] = tNow;
1781 m_values[nextFree] = valueNow;
1782 m_maxSize = std::max(m_maxSize, size());
1783 }
1784
1785 // Prepend an older entry to the beginning of the list. No cleanup is done.
1786 void prepend(double tNewOldest, const T& value) {
1787 assert(empty() || tNewOldest < m_times[m_oldest]);
1788 if (full()) makeMoreRoom();
1789 m_oldest = empty() ? 0 : getArrayIndex(-1);
1790 m_times[m_oldest] = tNewOldest;
1791 m_values[m_oldest] = value;
1792 ++m_size;
1793 m_maxSize = std::max(m_maxSize, size());
1794 }
1795
1796 // This is a specialized copy assignment for copying an old buffer
1797 // to a new one with updated contents. We are told the earliest time we'll
1798 // be asked about from now on, and won't copy any entries older than those
1799 // needed to answer that earliest request. We won't copy anything at or
1800 // newer than tNow, and finally we'll push (tNow,valueNow) as the newest
1801 // entry.
1802 void copyInAndUpdate(const Measure_Delay_Buffer& oldBuf, double tEarliest,
1803 double tNow, const T& valueNow) {
1804 // clear all current entries (no heap activity)
1805 m_oldest = m_size = 0;
1806
1807 // determine how may old entries we have to keep
1808 int firstNeeded = oldBuf.countNumUnneededOldEntries(tEarliest);
1809 int lastNeeded = oldBuf.findLastEarlier(tNow); // might be -1
1810 int numOldEntriesToKeep = lastNeeded-firstNeeded+1;
1811 int newSize = numOldEntriesToKeep+1; // includes the new one
1812
1813 int newSizeRequest = -1;
1814 if (capacity() < newSize) {
1815 newSizeRequest = std::max((int)InitialAllocation,
1816 (int)GrowthFactor * newSize);
1817 ++m_nGrows;
1818 } else if (capacity() > std::max((int)MaxShrinkProofSize,
1819 (int)TooBigFactor * newSize)) {
1820 newSizeRequest = std::max((int)MaxShrinkProofSize,
1821 (int)GrowthFactor * newSize);
1822 ++m_nShrinks;
1823 }
1824
1825 // Reallocate space if advisable.
1826 if (newSizeRequest != -1) {
1827 const double dNaN = NTraits<double>::getNaN();
1828 m_values.resize(newSizeRequest);
1829 if (m_values.capacity() > m_values.size())
1830 m_values.resize(m_values.capacity()); // don't waste any
1831 m_times.resize(m_values.size(), dNaN);
1832 }
1833
1834 m_maxCapacity = std::max(m_maxCapacity, capacity());
1835
1836 // Copy the entries we need to keep.
1837 int nxt = 0;
1838 for (int i=firstNeeded; i<=lastNeeded; ++i, ++nxt) {
1839 m_times[nxt] = oldBuf.getEntryTime(i);
1840 m_values[nxt] = oldBuf.getEntryValue(i);
1841 }
1842 // Now add the newest entry and set the size.
1843 m_times[nxt] = tNow;
1844 m_values[nxt] = valueNow;
1845 assert(nxt+1==newSize);
1846 m_size = nxt+1;
1847 m_maxSize = std::max(m_maxSize, size());
1848 }

1850 // Given the current time and value and the earlier time at which the
1851 // value is needed, use the buffer and (if necessary) the current value
1852 // to estimate the delayed value.
1853 T calcValueAtTime(double tDelay, double tNow, const T& valueNow) const;
1854
1855 // Given the current time but *not* the current value of the source measure,
1856 // provide an estimate for the value at tDelay=tNow-delay using only the
1857 // buffer contents and linear interpolation or extrapolation.
1858 void calcValueAtTimeLinearOnly(double tDelay, T& delayedValue) const {
1859 if (empty()) {
1860 // Nothing in the buffer?? Shouldn't happen. Return empty Vector
1861 // or NaN for fixed-size types.
1862 Measure_Num<T>::makeNaNLike(T(), delayedValue);
1863 return;
1864 }
1865
1866 int firstLater = findFirstLaterOrEq(tDelay);
1867
1868 if (firstLater > 0) {
1869 // Normal case: tDelay is between two buffer entries.
1870 int firstEarlier = firstLater-1;
1871 double t0=getEntryTime(firstEarlier), t1=getEntryTime(firstLater);
1872 const T& v0=getEntryValue(firstEarlier);
1873 const T& v1=getEntryValue(firstLater);
1874 Real fraction = Real((tDelay-t0)/(t1-t0));
1875 delayedValue = T(v0 + fraction*(v1-v0));
1876 return;
1877 }
1878
1879 if (firstLater==0) {
1880 // Startup case: tDelay is at or before the oldest buffer entry.
1881 // Assume the value was flat before that.
1882 delayedValue = getEntryValue(firstLater);
1883 return;
1884 }
1885
1886 // tDelay is later than the latest entry in the buffer. We are going
1887 // to have to extrapolate (yuck).
1888
1889 if (size() == 1) {
1890 // Just one entry; we'll have to assume the value is flat.
1891 delayedValue = getEntryValue(0);
1892 return;
1893 }
1894
1895 // Extrapolate using the last two entries.
1896 double t0=getEntryTime(size()-2), t1=getEntryTime(size()-1);
1897 const T& v0=getEntryValue(size()-2);
1898 const T& v1=getEntryValue(size()-1);
1899 Real fraction = Real((tDelay-t0)/(t1-t0)); // > 1
1900 assert(fraction > 1.0);
1901 delayedValue = T(v0 + fraction*(v1-v0)); // Extrapolate.
1902 }

1904 // Return the number of times we had to grow the buffer.
1905 int getNumGrows() const {return m_nGrows;}
1906 // Return the number of times we decided the buffer was so overallocated
1907 // that we had to shrink it.
1908 int getNumShrinks() const {return m_nShrinks;}
1909 // Return the largest number of values we ever had in the buffer.
1910 int getMaxSize() const {return m_maxSize;}
1911 // Return the largest capacity the buffer ever had.
1912 int getMaxCapacity() const {return m_maxCapacity;}

1914private:
1915 // Return the i'th oldest entry
1916 // (0 -> oldest, size-1 -> newest, size -> first free, -1 -> last free)
1917 int getArrayIndex(int i) const
1918 { assert(-1<=i && i<=size());
1919 const int rawIndex = m_oldest + i;
1920 if (rawIndex < 0) return rawIndex + capacity();
1921 else return rawIndex % capacity(); }
1922
1923 // Remove all but two entries older than the given time.
1924 void forgetEntriesMuchOlderThan(double tEarliest) {
1925 const int numToRemove = countNumUnneededOldEntries(tEarliest);
1926 if (numToRemove) {
1927 m_oldest = getArrayIndex(numToRemove);
1928 m_size -= numToRemove;
1929 }
1930 }
1931
1932 // Count up how many old entries at the beginning of the buffer are so old
1933 // that they wouldn't be needed to respond to a request at time tEarliest or
1934 // later. We'll keep no more than two entries earlier than tEarliest.
1935 int countNumUnneededOldEntries(double tEarliest) const {
1936 const int firstLater = findFirstLaterOrEq(tEarliest);
1937 return std::max(0, firstLater-2);
1938 }
1939
1940 // Given the time now, delete anything at the end of the queue that is
1941 // at that same time or later.
1942 void removeEntriesLaterOrEq(double t) {
1943 int lastEarlier = findLastEarlier(t);
1944 m_size = lastEarlier+1;
1945 if (m_size==0) m_oldest=0; // restart at beginning of array
1946 }
1947
1948 // Return the entry number (0..size-1) of the first entry whose time
1949 // is >= the given time, or -1 if there is none such.
1950 int findFirstLaterOrEq(double tDelay) const {
1951 for (int i=0; i < size(); ++i)
1952 if (getEntryTime(i) >= tDelay)
1953 return i;
1954 return -1;
1955 }
1956
1957 // Return the entry number(size-1..0) of the last entry whose time
1958 // is < the given time, or -1 if there is none such.
1959 int findLastEarlier(double t) const {
1960 for (int i=size()-1; i>=0; --i)
1961 if (getEntryTime(i) < t)
1962 return i;
1963 return -1;
1964 }
1965
1966 // We don't have enough space. This is either the initial allocation or
1967 // we need to double the current space.
1968 void makeMoreRoom() {
1969 const int newSizeRequest = std::max((int)InitialAllocation,
1970 (int)GrowthFactor * size());
1971 resize(newSizeRequest);
1972 ++m_nGrows;
1973 m_maxCapacity = std::max(m_maxCapacity, capacity());
1974 }
1975
1976 // We are wasting a lot of space, reduce the heap allocation to just
1977 // double what we're using now.
1978 void makeLessRoom() {
1979 const int targetMaxSize = std::max((int)MaxShrinkProofSize,
1980 (int)GrowthFactor * size());
1981 if (capacity() > targetMaxSize) {
1982 resize(targetMaxSize);
1983 ++m_nShrinks;
1984 }
1985 }
1986
1987 // Reallocate memory to get more space or stop wasting space. The new
1988 // size request must be big enough to hold all the current contents. The
1989 // amount we actually get may be somewhat larger than the request. On
1990 // return, the times and values arrays will have been resized and the
1991 // oldest entry will now be entry 0.
1992 void resize(int newSizeRequest) {
1993 assert(newSizeRequest >= size());
1994 const double dNaN = NTraits<double>::getNaN();
1995 Array_<T,int> newValues(newSizeRequest);
1996 if (newValues.capacity() > newValues.size())
1997 newValues.resize(newValues.capacity()); // don't waste any
1998 Array_<double,int> newTimes(newValues.size(), dNaN);
1999
2000 // Pack existing values into start of new arrays.
2001 for (int i=0; i < size(); ++i) {
2002 const int ix = getArrayIndex(i);
2003 newTimes[i] = m_times[ix];
2004 newValues[i] = m_values[ix];
2005 }
2006 m_times.swap(newTimes); // switch heap space
2007 m_values.swap(newValues);
2008 m_oldest = 0; // starts at the beginning now; size unchanged
2009 }
2010
2011 // Initialize everything to its default-constructed state.
2012 void initDataMembers() {
2013 m_times.clear(); m_values.clear();
2014 m_oldest=m_size=0;
2015 m_nGrows=m_nShrinks=m_maxSize=m_maxCapacity=0;
2016 }
2017
2018 // These are circular buffers of the same size.
2019 Array_<double,int> m_times;
2020 Array_<T,int> m_values;
2021 int m_oldest; // Array index of oldest (time,value)
2022 int m_size; // number of entries in use
2023
2024 // Statistics.
2025 int m_nGrows, m_nShrinks, m_maxSize, m_maxCapacity;
