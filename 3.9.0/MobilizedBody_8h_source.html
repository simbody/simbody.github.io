761SpatialVec findBodyVelocityInAnotherBody
762 (const State& state, const MobilizedBody& inBodyA) const
763{
764 const SpatialVec& V_GB = this->getBodyVelocity(state);
765 const SpatialVec& V_GA = inBodyA.getBodyVelocity(state);
766 // angular velocity of B in A, exp in G
767 const Vec3 w_AB_G = V_GB[0]-V_GA[0]; // 3 flops
768
769 // Angular vel. was easy, but for linear vel. we have to add in an wXr term.
770
771 const Transform& X_GB = getBodyTransform(state);
772 const Transform& X_GA = inBodyA.getBodyTransform(state);
773 // vector from Ao to Bo, exp in G
774 const Vec3 p_AB_G = X_GB.p() - X_GA.p(); // 3 flops
775 // d/dt p taken in G
776 const Vec3 p_AB_G_dot = V_GB[1] - V_GA[1]; // 3 flops
777
778 // d/dt p taken in A, exp in G
779 const Vec3 v_AB_G = p_AB_G_dot - V_GA[0] % p_AB_G; // 12 flops
780
781 // We're done, but answer is expressed in Ground. Reexpress in A and return.
782 return ~X_GA.R()*SpatialVec(w_AB_G, v_AB_G); // 30 flops
783}

839Vec3 findBodyAngularAccelerationInAnotherBody
840 (const State& state, const MobilizedBody& inBodyA) const
841{
842 const Rotation& R_GA = inBodyA.getBodyRotation(state);
843 const Vec3& w_GA = inBodyA.getBodyAngularVelocity(state);
844 const Vec3& w_GB = this->getBodyAngularVelocity(state);
845 const Vec3& b_GA = inBodyA.getBodyAngularAcceleration(state);
846 const Vec3& b_GB = this->getBodyAngularAcceleration(state);
847
848 // relative ang. vel. of B in A, exp. in G
849 const Vec3 w_AB_G = w_GB - w_GA; // 3 flops
850 const Vec3 w_AB_G_dot = b_GB - b_GA; // d/dt of w_AB_G taken in G; 3 flops
851
852 // We have the derivative in G; change it to derivative in A by adding
853 // in contribution caused by motion of G in A, that is w_AG X w_AB_G.
854 // (Note that w_AG=-w_GA.)
855 const Vec3 b_AB_G = w_AB_G_dot - w_GA % w_AB_G; // ang. accel. of B in A
856 // 12 flops
857
858 return ~R_GA * b_AB_G; // taken in A, expressed in A; 15 flops
859}

1033 (const State& state, const Vec3& locationOnB,

1241Inertia calcBodyInertiaAboutAnotherBodyStation
1242 (const State& state, const MobilizedBody& inBodyA,
1243 const Vec3& aboutLocationOnBodyA) const
1244{
1245 // get B's mass props MB, measured about Bo, exp. in B
1246 const MassProperties& MB_Bo_B = getBodyMassProperties(state);
1247
1248 // Calculate the vector from the body B origin (current "about" point) to
1249 // the new "about" point PA, expressed in B.
1250 const Vec3 p_Bo_PA =
1251 findStationAtAnotherBodyStation(state, inBodyA, aboutLocationOnBodyA);
1252
1253 // Now shift the "about" point for body B's inertia IB to PA, but still
1254 // expressed in B.
1255 const Inertia IB_PA_B = MB_Bo_B.calcShiftedInertia(p_Bo_PA);
1256
1257 // Finally reexpress the inertia in the A frame.
1258 const Rotation R_BA =
1259 inBodyA.findBodyRotationInAnotherBody(state, *this);
1260 const Inertia IB_PA_A = IB_PA_B.reexpress(R_BA);
1261 return IB_PA_A;
1262}

1295Real calcStationToStationDistance(const State& state,
1296 const Vec3& locationOnBodyB,
1297 const MobilizedBody& bodyA,
1298 const Vec3& locationOnBodyA) const
1299{
1300 if (isSameMobilizedBody(bodyA))

1304 this->findStationLocationInGround(state,locationOnBodyB);

1315Real calcStationToStationDistanceTimeDerivative
1316 (const State& state,
1317 const Vec3& locationOnBodyB,
1318 const MobilizedBody& bodyA,
1319 const Vec3& locationOnBodyA) const
1320{
1321 if (isSameMobilizedBody(bodyA))
1322 return 0;
1323
1324 Vec3 rB, rA, vB, vA;
1325 this->findStationLocationAndVelocityInGround(state,locationOnBodyB,rB,vB);
1326 bodyA.findStationLocationAndVelocityInGround(state,locationOnBodyA,rA,vA);
1327 const Vec3 r = rA-rB, v = vA-vB;
1328 const Real d = r.norm();
1329
1330 // When the points are coincident, the rate of change of distance is just
1331 // their relative speed. Otherwise, it is the speed along the direction of
1332 // separation.

1343Real calcStationToStationDistance2ndTimeDerivative
1344 (const State& state,
1345 const Vec3& locationOnBodyB,
1346 const MobilizedBody& bodyA,
1347 const Vec3& locationOnBodyA) const
1348{
1349 if (isSameMobilizedBody(bodyA))
1350 return 0;
1351
1352 Vec3 rB, rA, vB, vA, aB, aA;
1353 this->findStationLocationVelocityAndAccelerationInGround
1354 (state,locationOnBodyB,rB,vB,aB);
1355 bodyA.findStationLocationVelocityAndAccelerationInGround
1356 (state,locationOnBodyA,rA,vA,aA);
1357
1358 const Vec3 r = rA-rB, v = vA-vB, a = aA-aB;
1359 const Real d = r.norm();
1360
1361 // This method is the time derivative of
1362 // calcFixedPointToPointDistanceTimeDerivative(), so it must follow the same
1363 // two cases. That is, when the points are coincident the change in
1364 // separation rate is the time derivative of the speed |v|, otherwise it is
1365 // the time derivative of the speed along the separation vector.
1366
1367 if (d==0) {
1368 // Return d/dt |v|. This has two cases: if |v| is zero, the rate of
1369 // change of speed is just the points' relative acceleration magnitude.
1370 // Otherwise, it is the acceleration in the direction of the current
1371 // relative velocity vector.
1372 const Real s = v.norm(); // speed
1373 if (s==0) return a.norm();
1374 else return dot(a, v/s);
1375 }
1376
1377 // Points are separated.
1378 const Vec3 u = r/d; // u is separation direction (a unit vector from B to A)
1379 const Vec3 vp = v - dot(v,u)*u; // velocity perp. to separation direction
1380 return dot(a,u) + dot(vp,v)/d;
1381}
