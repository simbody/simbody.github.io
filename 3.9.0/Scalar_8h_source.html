361// It is safer to templatize using complex classes, and doesn't make
362// debugging any worse since complex is already templatized.
363// 5 flops vs. 6 for general complex multiply.
364template <class P> inline
365std::complex<P> square(const std::complex<P>& x) {
366 const P re=x.real(), im=x.imag();
367 return std::complex<P>(re*re-im*im, 2*re*im);
368}

370// We can square a conjugate and clean it up back to complex at
371// the same time at zero cost (or maybe 1 negation depending
372// on how the compiler handles the "-2" below).
373template <class P> inline
374std::complex<P> square(const conjugate<P>& x) {
375 const P re=x.real(), nim=x.negImag();
376 return std::complex<P>(re*re-nim*nim, -2*re*nim);
377}

427// To keep this cheap we'll defer getting rid of the negator<> until
428// some other operation. We cube -x and then recast that to negator<>
429// on return, for a total cost of 2 flops.
430inline negator<float> cube(const negator<float>& x) {
431 return negator<float>::recast(cube(-x));
432}

434 return negator<double>::recast(cube(-x));

437// Cubing a complex this way is cheaper than doing it by
438// multiplication. Cost here is 8 flops vs. 11 for a square
439// followed by a multiply.
440template <class P> inline
441std::complex<P> cube(const std::complex<P>& x) {
442 const P re=x.real(), im=x.imag(), rr=re*re, ii=im*im;
443 return std::complex<P>(re*(rr-3*ii), im*(3*rr-ii));
444}
445
446// Cubing a negated complex allows us to cube and eliminate the
447// negator at the same time for zero extra cost. Compare the
448// expressions here to the normal cube above to see the free
449// sign changes in both parts. 8 flops here.
450template <class P> inline
451std::complex<P> cube(const negator< std::complex<P> >& x) {
452 // -x is free for a negator
453 const P nre=(-x).real(), nim=(-x).imag(), rr=nre*nre, ii=nim*nim;
454 return std::complex<P>(nre*(3*ii-rr), nim*(ii-3*rr));
455}

457// Cubing a conjugate this way saves a lot over multiplying, and
458// also lets us convert the result to complex for free.
459template <class P> inline
460std::complex<P> cube(const conjugate<P>& x) {
461 const P re=x.real(), nim=x.negImag(), rr=re*re, ii=nim*nim;
462 return std::complex<P>(re*(rr-3*ii), nim*(ii-3*rr));
463}

466// Cubing a negated conjugate this way saves a lot over multiplying, and
467// also lets us convert the result to non-negated complex for free.
468template <class P> inline
469std::complex<P> cube(const negator< conjugate<P> >& x) {
470 // -x is free for a negator
471 const P nre=(-x).real(), im=(-x).negImag(), rr=nre*nre, ii=im*im;
472 return std::complex<P>(nre*(3*ii-rr), im*(3*rr-ii));
473}

1182static inline std::pair<T,T> completeEllipticIntegralsKE_T(T m) {
1183 const T SignificantReal = NTraits<T>::getSignificant();
1184 const T TenEps = 10*NTraits<T>::getEps();
1185 const T Infinity = NTraits<T>::getInfinity();
1186 const T PiOver2 = NTraits<T>::getPi() / 2;
1187
1188 // Allow a little slop in the argument since it may have resulted
1189 // from a numerical operation that gave 0 or 1 plus or minus
1190 // roundoff noise.
1191 SimTK_ERRCHK1_ALWAYS(-SignificantReal < m && m < 1+SignificantReal,
1192 "completeEllipticIntegralsKE()",
1193 "Argument m (%g) is outside the legal range [0,1].", (double)m);
1194 if (m >= 1) return std::make_pair(Infinity, (T)1);
1195 if (m <= 0) return std::make_pair(PiOver2, PiOver2);
1196
1197 const T k = std::sqrt(1-m); // ~25 flops
1198 T v1=1, w1=k, c1=1, d1=k*k; // initialize iteration
1199 do { // ~50 flops per iteration
1200 T v2 = (v1+w1)/2;
1201 T w2 = std::sqrt(v1*w1);
1202 T c2 = (c1+d1)/2;
1203 T d2 = (w1*c1+v1*d1)/(v1+w1);
1204 v1=v2; w1=w2; c1=c2; d1=d2;
1205 } while(std::abs(v1-w1) >= TenEps);
1206
1207 const T K = PiOver2/v1; // ~20 flops
1208 const T E = K*c1;
1209
1210 return std::make_pair(K,E);
1211}
