184 typename CNT<ScalarNormSq>::TSqrt
185 normRMS(int* worstOne=0) const {
186 if (!CNT<ELT>::IsScalar)
187 SimTK_THROW1(Exception::Cant,
188 "Vector::normRMS() only defined for scalar elements.");
189 const int n = nrow();
190 if (n == 0) {
191 if (worstOne) *worstOne = -1;
192 return typename CNT<ScalarNormSq>::TSqrt(0);
193 }
194
195 ScalarNormSq sumsq = 0;
196 if (worstOne) {
197 *worstOne = 0;
198 ScalarNormSq maxsq = 0;
199 for (int i=0; i<n; ++i) {
200 const ScalarNormSq v2 = square((*this)[i]);
201 if (v2 > maxsq) maxsq=v2, *worstOne=i;
202 sumsq += v2;
203 }
204 } else { // don't track the worst element
205 for (int i=0; i<n; ++i) {
206 const ScalarNormSq v2 = square((*this)[i]);
207 sumsq += v2;
208 }
209 }
210
211 return CNT<ScalarNormSq>::sqrt(sumsq/n);
212 }

220 typename CNT<ScalarNormSq>::TSqrt
221 weightedNormRMS(const VectorBase<EE>& w, int* worstOne=0) const {
222 if (!CNT<ELT>::IsScalar || !CNT<EE>::IsScalar)
223 SimTK_THROW1(Exception::Cant,
224 "Vector::weightedNormRMS() only defined for scalar elements"
225 " and weights.");
226 const int n = nrow();
227 assert(w.nrow()==n);
228 if (n == 0) {
229 if (worstOne) *worstOne = -1;
230 return typename CNT<ScalarNormSq>::TSqrt(0);
231 }
232
233 ScalarNormSq sumsq = 0;
234 if (worstOne) {
235 *worstOne = 0;
236 ScalarNormSq maxsq = 0;
237 for (int i=0; i<n; ++i) {
238 const ScalarNormSq wv2 = square(w[i]*(*this)[i]);
239 if (wv2 > maxsq) maxsq=wv2, *worstOne=i;
240 sumsq += wv2;
241 }

244 const ScalarNormSq wv2 = square(w[i]*(*this)[i]);
245 sumsq += wv2;
