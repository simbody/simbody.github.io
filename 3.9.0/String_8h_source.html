375// Implementation of the generic templatized String constructor.
376template <class T> inline
377String::String(const T& t) {
378 std::ostringstream os;
379 *this = stringStreamInsertHelper(os, t, true).str();
380}

382// Implementation of the generic templatized String constructor
383// with precision.
384// To ensure consistent behavior of ostream::setprecision() across operating
385// systems and whether a build is debug or release, precision is bounded:
386// 1 <= precision <= SimTK::LosslessNumDigitsReal
387template <class T> inline
388String::String(const T& t, int precision) {
389 std::ostringstream os;
390 if(precision < 1)
391 precision = 1;
392 else if(precision > SimTK::LosslessNumDigitsReal)
393 precision = SimTK::LosslessNumDigitsReal;
394 os << std::setprecision(precision);
395 *this = stringStreamInsertHelper(os, t, true).str();
396}

399// This namespace-level static method should not be necessary but gcc 4.1
400// still has trouble with template specialization for template member
401// functions. So rather than specializing the tryConvertTo() member, I'm
402// specializing this helper function instead.
403template <class T> inline static
404bool tryConvertStringTo(const String& value, T& out) {
405 std::istringstream sstream(value);
406 stringStreamExtractHelper(sstream, out, true);
407 if (sstream.fail()) return false;
408 if (sstream.eof()) return true;
409 // Successful conversion but didn't use all the characters. Maybe the
410 // rest is just whitespace?
411 std::ws(sstream); // Skip trailing whitespace if any.
412 return sstream.eof(); // We must have used up the whole string now.
413}

477template <class T> inline void
478String::convertTo(T& out) const {
479 const int MaxStr = 50;
480 const bool convertOK = tryConvertTo<T>(out);
481 if (convertOK) return;
482
483 // Make sure we don't try to output more than MaxStr characters of
484 // the bad string in the error message.
485 String shorter = this->substr(0, MaxStr);
486 if (shorter.size() < this->size()) shorter += " ...";
487 SimTK_ERRCHK2_ALWAYS(convertOK, "String::convertTo()",
488 "Couldn't interpret string '%s' as type T=%s.",
489 shorter.c_str(), NiceTypeName<T>::namestr().c_str());
490}
