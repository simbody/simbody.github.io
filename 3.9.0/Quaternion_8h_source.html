174 Quaternion_<P> multiply(const Quaternion_& q2) const {
175 const Quaternion_& q1 = *this;
176 const double w1 = q1[0], x1 = q1[1], y1 = q1[2], z1 = q1[3];
177 const double w2 = q2[0], x2 = q2[1], y2 = q2[2], z2 = q2[3];
178
179 // Compute the Hamilton product
180 const double w = w1 * w2 - x1 * x2 - y1 * y2 - z1 * z2;
181 const double x = w1 * x2 + x1 * w2 + y1 * z2 - z1 * y2;
182 const double y = w1 * y2 - x1 * z2 + y1 * w2 + z1 * x2;
183 const double z = w1 * z2 + x1 * y2 - y1 * x2 + z1 * w2;
184
185 return Quaternion_<P>(w, x, y, z);
186 }
