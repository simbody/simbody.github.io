342void setRotationToBodyFixedXYZ(const Vec3P& c, const Vec3P& s) {
343 Mat33P& R = *this;
344 const RealP s0s1=s[0]*s[1], s2c0=s[2]*c[0], c0c2=c[0]*c[2], nc1= -c[1];
345
346 R = Mat33P( c[1]*c[2] , s[2]*nc1 , s[1] ,
347 s2c0 + s0s1*c[2] , c0c2 - s0s1*s[2] , s[0]*nc1 ,
348 s[0]*s[2] - s[1]*c0c2 , s[0]*c[2] + s[1]*s2c0 , c[0]*c[1] );
349}

386Rotation_ mirror(const CoordinateAxis& axis) const {
387 const int k = axis;
388 Mat33P R = asMat33();
389 // Element (i,j) is scaled by s[i]*s[j], so this negates the row and column
390 // for the mirrored axis k. The sign of the diagonal element (k,k) remains
391 // unchanged.
392 for (int i = 0; i < 3; ++i)
393 if (i != k) { R(k,i) = -R(k,i); R(i,k) = -R(i,k); }
394 return Rotation_(R, true);
395}

413static Vec3P multiplyByBodyXYZ_N_P(const Vec2P& cosxy,
414 const Vec2P& sinxy,
415 RealP oocosy,
416 const Vec3P& w_PB)
417{
418 const RealP s0 = sinxy[0], c0 = cosxy[0];
419 const RealP s1 = sinxy[1];
420 const RealP w0 = w_PB[0], w1 = w_PB[1], w2 = w_PB[2];
421
422 const RealP t = (s0*w1-c0*w2)*oocosy;
423 return Vec3P( w0 + t*s1, c0*w1 + s0*w2, -t ); // qdot
424}

429static Vec3P multiplyByBodyXYZ_NT_P(const Vec2P& cosxy,
430 const Vec2P& sinxy,
431 RealP oocosy,
432 const Vec3P& q)
433{
434 const RealP s0 = sinxy[0], c0 = cosxy[0];
435 const RealP s1 = sinxy[1];

612static Mat33P calcNDotForBodyXYZInBodyFrame
613 (const Vec3P& cq, const Vec3P& sq, const Vec3P& qdot)
614{
615 const RealP s1 = sq[1], c1 = cq[1];
616 const RealP s2 = sq[2], c2 = cq[2];
617 const RealP ooc1 = 1/c1;
618 const RealP s2oc1 = s2*ooc1, c2oc1 = c2*ooc1;
619
620 const RealP t = qdot[1]*s1*ooc1;
621 const RealP a = t*s2oc1 + qdot[2]*c2oc1; // d/dt s2oc1
622 const RealP b = t*c2oc1 - qdot[2]*s2oc1; // d/dt c2oc1
623
624 return Mat33P( b , -a , 0,
625 qdot[2]*c2 , -qdot[2]*s2 , 0,
626 -(s1*b + qdot[1]*c2) , s1*a + qdot[1]*s2 , 0 );
627}

869static Vec3P convertAngVelToBodyFixed321Dot(const Vec3P& q, const Vec3P& w_PB_B) {
870 const RealP s1 = std::sin(q[1]), c1 = std::cos(q[1]);
871 const RealP s2 = std::sin(q[2]), c2 = std::cos(q[2]);
872 const RealP ooc1 = RealP(1)/c1;
873 const RealP s2oc1 = s2*ooc1, c2oc1 = c2*ooc1;
874
875 const Mat33P E( 0, s2oc1 , c2oc1 ,
876 0, c2 , -s2 ,
877 1, s1*s2oc1 , s1*c2oc1 );
878 return E * w_PB_B;
