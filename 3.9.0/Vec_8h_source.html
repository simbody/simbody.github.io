60 result[N-1] = r1[N-1] + r2[N-1];

1428template <int M, class E, int S, class CHAR, class TRAITS> inline
1429std::basic_istream<CHAR,TRAITS>&
1430operator>>(std::basic_istream<CHAR,TRAITS>& is, Vec<M,E,S>& v) {
1431 CHAR tilde;
1432 is >> tilde; if (is.fail()) return is;
1433 if (tilde != CHAR('~')) {
1434 tilde = CHAR(0);
1435 is.unget(); if (is.fail()) return is;
1436 }
1437
1438 CHAR openBracket, closeBracket;
1439 is >> openBracket; if (is.fail()) return is;
1440 if (openBracket==CHAR('('))
1441 closeBracket = CHAR(')');
1442 else if (openBracket==CHAR('['))
1443 closeBracket = CHAR(']');
1444 else {
1445 closeBracket = CHAR(0);
1446 is.unget(); if (is.fail()) return is;
1447 }
1448
1449 // If we saw a "~" but then we didn't see any brackets, that's an
1450 // error. Set the fail bit and return.
1451 if (tilde != CHAR(0) && closeBracket == CHAR(0)) {
1452 is.setstate( std::ios::failbit );
1453 return is;
1454 }
1455
1456 for (int i=0; i < M; ++i) {
1457 is >> v[i];
1458 if (is.fail()) return is;
1459 if (i != M-1) {
1460 CHAR c; is >> c; if (is.fail()) return is;
1461 if (c != ',') is.unget();
1462 if (is.fail()) return is;
1463 }
1464 }
1465
1466 // Get the closing bracket if there was an opening one. If we don't
1467 // see the expected character we'll set the fail bit in the istream.
1468 if (closeBracket != CHAR(0)) {
1469 CHAR closer; is >> closer; if (is.fail()) return is;
1470 if (closer != closeBracket) {
1471 is.unget(); if (is.fail()) return is;
1472 is.setstate( std::ios::failbit );
1473 }
1474 }
1475
1476 return is;
1477}
