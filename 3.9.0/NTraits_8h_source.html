64// Consteval square root. Uses `std::sqrt` when `__cpp_lib_constexpr_cmath` is
65// available (C++23). Otherwise uses a Newton-Raphson solver, which may be
66// off by 1 ULP due to differences in rounding.
67consteval double consteval_sqrt(double x) noexcept {
68#if defined(__cpp_lib_constexpr_cmath) && __cpp_lib_constexpr_cmath >= 202202L
69 return std::sqrt(x); // In new C++es, just call `std::sqrt`.
70#else
71 static_assert(std::numeric_limits<double>::is_iec559);
72
73 constexpr double max_double = std::numeric_limits<double>::max();
74 constexpr double nan = std::numeric_limits<double>::quiet_NaN();
75
76 // Trivial/edge cases.
77 if (x != x) return x; // forward NaNs
78 if (x < 0.0) return nan; // sqrt(negative_number) -> NaN
79 if (x == 0.0) return x; // preserve -0 and +0
80 if (x > max_double) return x; // +inf
81
82 // Subnormal scaling.
83 double scale = 1.0;
84 if (x < std::numeric_limits<double>::min()) {
85 x *= 4503599627370496.0; // 2^52
86 scale = 1.0 / 67108864.0; // 2^-26
87 }
88
89 auto b = std::bit_cast<std::uint64_t>(x);
90 b = (b + (1023ULL << 52)) >> 1;
91 auto g = std::bit_cast<double>(b);
92
93 // Newton-Raphson solver: might be off by 1ULP
94 for (;;) {
95 const auto next = (g + x / g) * 0.5;
96 if (next >= g) break;
97 g = next;
98 }
99
100 return g * scale;
101#endif
102}

107// This is the 3rd type of number, conjugate. It is like complex except that
108// the represented value is the conjugate of the value represented by a complex
109// number containing the same bit pattern. That is, complex and conjugate both
110// contain two real numbers, re and im, with complex(re,im) meaning
111// (re + im*i) while conjugate(re,im) means (re - im*i). It is guaranteed that
112// our conjugate type and complex type have identical sizes and representations.
113// Together, these defininitions and guarantees permit conjugation
114// to be done by reinterpretation rather than be computation.

129// This is an adaptor for numeric types which negates the apparent values. A
130// negator<N> has exactly the same internal representation as a numeric value N,
131// but it is to be interpreted has having the negative of the value it would
132// have if interpreted as an N. This permits negation to be done by
133// reinterpretation rather than computation. A full set of arithmetic operators
134// are provided involving negator<N>'s and N's. Sometimes we can save an op or
135// two this way. For example negator<N>*negator<N> can be performed as an N*N
136// since the negations cancel, and we saved two floating point negations.
